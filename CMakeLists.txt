cmake_minimum_required(VERSION 3.10)
project(c99gui VERSION 2.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

option(C99GUI_HEADLESS "Build in headless mode (no windowing dependencies)" OFF)
option(C99GUI_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" ON)

# TUI - no graphics dependencies
add_executable(c99gui-tui src/tui/c99gui.c)
if(MSVC)
  target_compile_options(c99gui-tui PRIVATE /W4)
  if(C99GUI_WARNINGS_AS_ERRORS)
    target_compile_options(c99gui-tui PRIVATE /WX)
  endif()
  set_property(TARGET c99gui-tui PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
else()
  target_compile_options(c99gui-tui PRIVATE -Wall -Wextra -Wpedantic)
  if(C99GUI_WARNINGS_AS_ERRORS)
    target_compile_options(c99gui-tui PRIVATE -Werror)
  endif()
endif()

# Helper function to configure GUI executables
function(configure_gui_exe TARGET_NAME SOURCE_FILE)
  add_executable(${TARGET_NAME} ${SOURCE_FILE})
  
  if(MSVC)
    target_compile_options(${TARGET_NAME} PRIVATE /W4 /utf-8)
    if(C99GUI_WARNINGS_AS_ERRORS)
      target_compile_options(${TARGET_NAME} PRIVATE /WX)
    endif()
    set_property(TARGET ${TARGET_NAME} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_link_libraries(${TARGET_NAME} PRIVATE user32 gdi32)
  elseif(WIN32 OR CYGWIN)
    target_compile_options(${TARGET_NAME} PRIVATE -Wall -Wextra -Wpedantic)
    if(C99GUI_WARNINGS_AS_ERRORS)
      target_compile_options(${TARGET_NAME} PRIVATE -Werror)
    endif()
    target_link_libraries(${TARGET_NAME} PRIVATE user32 gdi32)
  elseif(APPLE)
    target_compile_options(${TARGET_NAME} PRIVATE -Wall -Wextra -Wpedantic)
    if(C99GUI_WARNINGS_AS_ERRORS)
      target_compile_options(${TARGET_NAME} PRIVATE -Werror)
    endif()
    find_library(COCOA_LIB Cocoa REQUIRED)
    find_library(CG_LIB CoreGraphics REQUIRED)
    target_link_libraries(${TARGET_NAME} PRIVATE ${COCOA_LIB} ${CG_LIB})
  else()
    # Linux / BSD / Unix
    target_compile_options(${TARGET_NAME} PRIVATE -Wall -Wextra -Wpedantic)
    if(C99GUI_WARNINGS_AS_ERRORS)
      target_compile_options(${TARGET_NAME} PRIVATE -Werror)
    endif()
    find_package(X11)
    if(X11_FOUND AND NOT C99GUI_HEADLESS)
      target_include_directories(${TARGET_NAME} PRIVATE ${X11_INCLUDE_DIR})
      target_link_libraries(${TARGET_NAME} PRIVATE ${X11_LIBRARIES})
    else()
      target_compile_definitions(${TARGET_NAME} PRIVATE C99GUI_HEADLESS)
    endif()
  endif()
  
  if(C99GUI_HEADLESS)
    target_compile_definitions(${TARGET_NAME} PRIVATE C99GUI_HEADLESS)
  endif()
endfunction()

# GUI programs
configure_gui_exe(c99gui-pixel src/pixel/gui.c)
configure_gui_exe(c99gui-vector src/vector/gui.c)
configure_gui_exe(c99gui-suite src/suite/gui.c)

include(CTest)
if(BUILD_TESTING)
  # TUI tests
  add_test(NAME tui_version COMMAND c99gui-tui --version)
  add_test(NAME tui_help COMMAND c99gui-tui --help)
  
  find_program(C99GUI_SH sh NAMES sh bash)
  if(C99GUI_SH)
    add_test(NAME tui_smoke COMMAND ${C99GUI_SH} ${CMAKE_CURRENT_SOURCE_DIR}/tests/tui_smoke.sh $<TARGET_FILE:c99gui-tui>)
    add_test(NAME pixel_smoke COMMAND ${C99GUI_SH} ${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.sh $<TARGET_FILE:c99gui-pixel>)
    add_test(NAME vector_smoke COMMAND ${C99GUI_SH} ${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.sh $<TARGET_FILE:c99gui-vector>)
    add_test(NAME suite_smoke COMMAND ${C99GUI_SH} ${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.sh $<TARGET_FILE:c99gui-suite>)
  endif()
endif()

install(TARGETS c99gui-tui c99gui-pixel c99gui-vector c99gui-suite
        RUNTIME DESTINATION bin)
