cmake_minimum_required(VERSION 3.10)
project(c99gui VERSION 1.0.0 LANGUAGES C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

option(C99GUI_HEADLESS "Build in headless mode (no windowing dependencies)" OFF)
option(C99GUI_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" ON)

add_executable(gui src/gui.c)

if(MSVC)
  target_compile_options(gui PRIVATE /W4 /utf-8)
  if(C99GUI_WARNINGS_AS_ERRORS)
    target_compile_options(gui PRIVATE /WX)
  endif()
  set_property(TARGET gui PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
  target_link_libraries(gui PRIVATE user32 gdi32)
elseif(WIN32 OR CYGWIN)
  target_compile_options(gui PRIVATE -Wall -Wextra -Wpedantic)
  if(C99GUI_WARNINGS_AS_ERRORS)
    target_compile_options(gui PRIVATE -Werror)
  endif()
  target_link_libraries(gui PRIVATE user32 gdi32)
elseif(APPLE)
  target_compile_options(gui PRIVATE -Wall -Wextra -Wpedantic)
  if(C99GUI_WARNINGS_AS_ERRORS)
    target_compile_options(gui PRIVATE -Werror)
  endif()
  find_library(COCOA_LIB Cocoa REQUIRED)
  find_library(CG_LIB CoreGraphics REQUIRED)
  target_link_libraries(gui PRIVATE ${COCOA_LIB} ${CG_LIB})
else()
  # Linux / BSD / Unix
  target_compile_options(gui PRIVATE -Wall -Wextra -Wpedantic)
  if(C99GUI_WARNINGS_AS_ERRORS)
    target_compile_options(gui PRIVATE -Werror)
  endif()
  find_package(X11)
  if(X11_FOUND AND NOT C99GUI_HEADLESS)
    target_include_directories(gui PRIVATE ${X11_INCLUDE_DIR})
    target_link_libraries(gui PRIVATE ${X11_LIBRARIES})
  else()
    target_compile_definitions(gui PRIVATE C99GUI_HEADLESS)
  endif()
endif()

if(C99GUI_HEADLESS)
  target_compile_definitions(gui PRIVATE C99GUI_HEADLESS)
endif()

include(CTest)
find_program(C99GUI_SH sh)
if(BUILD_TESTING AND C99GUI_SH)
  add_test(NAME smoke COMMAND ${C99GUI_SH} ${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.sh $<TARGET_FILE:gui>)
endif()

install(TARGETS gui RUNTIME DESTINATION bin)
